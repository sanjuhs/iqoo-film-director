cmake_minimum_required(VERSION 3.22.1)
project(minifilm_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-O3)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
set(LLAMA_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../phone-director/app/src/main/cpp/llama.cpp")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${LLAMA_VENDOR}"
    OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE REV_RESULT)
if(NOT REV_RESULT EQUAL 0 OR NOT LLAMA_REV STREQUAL "11fe02151f79c41d0d4af7da708755d73b9c0da6")
    message(FATAL_ERROR "Reuse the existing pinned official llama.cpp checkout; see README.md")
endif()
add_subdirectory("${LLAMA_VENDOR}" "llama-vendor-build")
add_library(model_bench SHARED benchmark_jni.cpp)
target_include_directories(model_bench PRIVATE "${LLAMA_VENDOR}/include")
target_link_libraries(model_bench PRIVATE llama android log)
