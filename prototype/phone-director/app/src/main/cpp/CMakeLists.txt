cmake_minimum_required(VERSION 3.22.1)
project(minifilm_director LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# AGP's debug variant otherwise builds inference kernels without optimization.
# Keep debug symbols while optimizing the actual model computation.
add_compile_options(-O3)
set(DIRECTOR_PHONE_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "Architecture verified on Phone (3a); empty retains portable arm64 baseline")
if(DIRECTOR_PHONE_ARM_ARCH)
    set(GGML_CPU_ARM_ARCH "${DIRECTOR_PHONE_ARM_ARCH}" CACHE STRING "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
# Local ignored dependency: official MIT-licensed llama.cpp, fixed provenance.
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp"
    OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE REV_RESULT)
if(NOT REV_RESULT EQUAL 0 OR NOT LLAMA_REV STREQUAL "11fe02151f79c41d0d4af7da708755d73b9c0da6")
    message(FATAL_ERROR "Fetch official llama.cpp at pinned commit 11fe02151f79c41d0d4af7da708755d73b9c0da6; see docs/local-ai-evidence.md")
endif()
add_subdirectory(llama.cpp)
add_library(director_llm SHARED director_jni.cpp)
target_include_directories(director_llm PRIVATE llama.cpp/include)
if(DIRECTOR_PHONE_ARM_ARCH STREQUAL "armv8.2-a+dotprod+fp16")
    target_compile_definitions(director_llm PRIVATE DIRECTOR_REQUIRE_DOTPROD_FP16=1)
endif()
target_link_libraries(director_llm PRIVATE llama android log)

add_subdirectory(whisper)
